```typescript
// `[CITED: developer.mozilla.org Web/API/StorageManager/persist]`
export async function requestPersistence(): Promise<boolean> {
  if (typeof navigator === 'undefined' || !('storage' in navigator) || !('persist' in navigator.storage)) {
    return false;
  }
  const alreadyPersistent = await navigator.storage.persisted();
  if (alreadyPersistent) return true;
  return await navigator.storage.persist();
}

export function isInstalledPWA(): boolean {
  if (typeof window === 'undefined') return false;
  return (
    window.matchMedia('(display-mode: standalone)').matches ||
    // iOS Safari legacy
    (navigator as any).standalone === true
  );
}

export function isIOSSafari(): boolean {
  if (typeof navigator === 'undefined') return false;
  const ua = navigator.userAgent;
  const isIOS = /iPad|iPhone|iPod/.test(ua) && !(window as any).MSStream;
  const isSafari = /Safari/.test(ua) && !/CriOS|FxiOS|EdgiOS/.test(ua);
  return isIOS && isSafari;
}
```

### 3. Create `src/lib/notifications/scheduler.ts`

In-app timer — best-effort while the app is open (RESEARCH §Pitfall 2 honest scope). Reads `reminder_time` from Dexie on every tick (cheap — single `settings.get`). Fires at most once per local_date.

```typescript
import { db } from '$lib/db/local';
import { localDate } from '$lib/db/mutations';

let lastFired: string | null = null;  // local_date of last fire, to prevent re-firing on same day

async function tick(): Promise<void> {
  if (typeof Notification === 'undefined' || Notification.permission !== 'granted') return;
  const setting = await db.settings.get('reminder_time');
  const targetHHMM = setting?.value ?? '21:00';
  const now = new Date();
  const hh = String(now.getHours()).padStart(2, '0');
  const mm = String(now.getMinutes()).padStart(2, '0');
  if (`${hh}:${mm}` !== targetHHMM) return;

  const today = localDate();
  if (lastFired === today) return;
  lastFired = today;

  // Use service worker registration if available (so notification click routes through SW)
  try {
    const reg = await navigator.serviceWorker?.ready;
    if (reg && 'showNotification' in reg) {
      await reg.showNotification('MoodLog', {
        body: 'Log today — how was it?',
        tag: `moodlog-reminder-${today}`,
        data: { date: 'today' }
      });
    } else {
      // Fallback: plain Notification (no deep-link routing through SW; tapping just focuses the tab)
      new Notification('MoodLog', { body: 'Log today — how was it?' });
    }
  } catch {
    // Silent — best-effort (PITFALLS #12: do not log)
  }
}

export function startReminderScheduler(): { stop: () => void } {
  if (typeof window === 'undefined') return { stop: () => {} };  // SSR no-op
  // First tick immediately (covers "app opened exactly at 21:00")
  void tick();
  const id = window.setInterval(tick, 60_000);
  return { stop: () => window.clearInterval(id) };
}

export function stopReminderScheduler(handle: { stop: () => void }): void {
  handle.stop();
}

// Test-only helper (exported for unit testing)
export function __resetLastFiredForTests(): void {
  lastFired = null;
}
```

```typescript
/// <reference lib="webworker" />
// `[CITED: developer.mozilla.org Web/API/ServiceWorkerGlobalScope/notificationclick_event]`
// SvelteKit `injectManifest` strategy — this file is the SW entry.
declare const self: ServiceWorkerGlobalScope;

// Reserved for @vite-pwa/sveltekit injection — do not remove the marker comment below:
// self.__WB_MANIFEST

// ENT-07 / D-21 — deep-link notification click to the entry form with today's date.
self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  const url = '/?date=today';
  event.waitUntil(
    (async () => {
      const list = await self.clients.matchAll({ type: 'window', includeUncontrolled: true });
      const existing = list.find((c) => c.url.includes(self.location.origin));
      if (existing) {
        await existing.focus();
        if ('navigate' in existing) {
          // @ts-expect-error — navigate is WindowClient-only
          await existing.navigate(url);
        }
      } else {
        await self.clients.openWindow(url);
      }
    })()
  );
});

// Phase 1 has no push server — explicit no-op makes the attack surface obvious (RESEARCH §Security).
self.addEventListener('push', () => {
  // Intentionally empty.
});

// Extract the click handler for unit testing (see tests/sw.test.ts).
export async function handleNotificationClick(
  event: { notification: { close: () => void }; waitUntil: (p: Promise<unknown>) => void },
  clients: {
    matchAll: (opts: { type: 'window'; includeUncontrolled: boolean }) => Promise<
      Array<{ url: string; focus: () => Promise<void>; navigate?: (u: string) => Promise<void> }>
    >;
    openWindow: (u: string) => Promise<unknown>;
  },
  origin: string
): Promise<string> {
  event.notification.close();
  const url = '/?date=today';
  const list = await clients.matchAll({ type: 'window', includeUncontrolled: true });
  const existing = list.find((c) => c.url.includes(origin));
  if (existing) {
    await existing.focus();
    if (existing.navigate) await existing.navigate(url);
  } else {
    await clients.openWindow(url);
  }
  return url;
}
```

```svelte
  import '../app.css';
  import { Toaster } from '$lib/components/ui/sonner';
  import { onMount, onDestroy } from 'svelte';
  import { page } from '$app/stores';
  import { goto } from '$app/navigation';

  import { getSetting } from '$lib/db/queries';
  import { requestPersistence } from '$lib/pwa/persist';
  import { startReminderScheduler } from '$lib/notifications/scheduler';
  import InstallBanner from '$lib/components/InstallBanner.svelte';

  let { children } = $props();

  let schedulerHandle: { stop: () => void } | null = null;

  onMount(async () => {
    // D-16 — persist silently on EVERY open (PITFALLS #7 / Pitfall 1 — idempotent call)
    const persisted = await requestPersistence();
    await setSetting('storage_persistent', persisted ? 'true' : 'false');

    // D-15 — redirect to onboarding on first open
    const onboarded = await getSetting('onboarded');
    const currentPath = $page.url.pathname;
    if (!onboarded?.value && currentPath !== '/onboarding') {
      await goto('/onboarding');
      return;
    }

    // D-21 — notification deep-link handler (?date=today)
    const params = $page.url.searchParams;
    if (params.get('date') === 'today') {
      resetDraft(localDate());
      const url = new URL(window.location.href);
      url.searchParams.delete('date');
      history.replaceState({}, '', url.toString());
    }

    // ENT-07 — start the in-app reminder scheduler (best-effort while open)
    schedulerHandle = startReminderScheduler();
  });

  onDestroy(() => {
    if (schedulerHandle) schedulerHandle.stop();
  });
</script>
```

### 8. Manifest icons (MANDATORY — iOS Safari rejects install if icons are too small)

The `manifest.icons` array in `vite.config.ts` from Plan 01 is empty. iOS Safari requires valid icons at install time — tiny placeholder PNGs (e.g. 1×1) will cause install to fail or produce a broken Home Screen tile, breaking ENT-08 install quality.

**Required pixel dimensions (non-negotiable):**
- `static/manifest-icon-192.png` — exactly **192×192 px**
- `static/manifest-icon-512.png` — exactly **512×512 px**
- `static/apple-touch-icon.png` — exactly **180×180 px**

All three MUST be valid PNGs at the listed dimensions. **DO NOT commit 1×1 transparent PNGs.** If branded artwork is not yet ready, generate neutral solid-colour squares at the exact required dimensions using the UI-SPEC background hex `#FAFAF9` (stone-50). Any deterministic tool works:

Option A — ImageMagick (if installed):
```
magick -size 192x192 xc:'#FAFAF9' static/manifest-icon-192.png
magick -size 512x512 xc:'#FAFAF9' static/manifest-icon-512.png
magick -size 180x180 xc:'#FAFAF9' static/apple-touch-icon.png
```

Option B — Node one-liner using `sharp` (install as a devDependency if needed):
```
node -e "const s=require('sharp');[['192',192],['512',512],['180',180]].forEach(([n,d])=>s({create:{width:d,height:d,channels:4,background:'#FAFAF9'}}).png().toFile(n==='180'?'static/apple-touch-icon.png':'static/manifest-icon-'+n+'.png'))"
```

Option C — `@vite-pwa/assets-generator` (preferred when a brand SVG exists at `static/favicon.svg`):
```
npx --yes @vite-pwa/assets-generator --preset minimal static/favicon.svg
```
Then rename generated files to the exact paths above.
